cmake_minimum_required(VERSION 3.20)
project(KCFProjectManager VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(KCF_PM_GUI "Build the independent Qt6 GUI" ON)
find_package(nlohmann_json 3.2 REQUIRED)
find_package(Threads REQUIRED)
foreach(name runtime element_main supervisor_main)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/templates/${name}.txt")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/templates/${name}.txt" ${name})
endforeach()
configure_file(src/templates.hpp.in generated/templates.hpp @ONLY)
add_library(kcf_project src/project.cpp src/process.cpp src/build_settings.cpp src/custom_module.cpp)
target_compile_features(kcf_project PUBLIC cxx_std_17)
target_include_directories(kcf_project PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(kcf_project PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(kcf_project PRIVATE -Wall -Wextra -Wpedantic)
add_executable(kcf-project src/cli.cpp)
target_link_libraries(kcf-project PRIVATE kcf_project)
if(KCF_PM_GUI)
  find_package(Qt6 REQUIRED COMPONENTS Widgets)
  add_executable(kcf-project-manager src/gui.cpp)
  target_link_libraries(kcf-project-manager PRIVATE kcf_project Qt6::Widgets)
  install(TARGETS kcf-project-manager RUNTIME DESTINATION bin)
endif()
install(TARGETS kcf_project kcf-project EXPORT KCFProjectManagerTargets ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/KCFProjectManagerConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/KCFProjectManagerConfig.cmake
  INSTALL_DESTINATION lib/cmake/KCFProjectManager)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/KCFProjectManagerConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(EXPORT KCFProjectManagerTargets NAMESPACE KCFProjectManager:: DESTINATION lib/cmake/KCFProjectManager)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/KCFProjectManagerConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/KCFProjectManagerConfigVersion.cmake DESTINATION lib/cmake/KCFProjectManager)
install(DIRECTORY include/ DESTINATION include)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME module_paths COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/modules.py $<TARGET_FILE:kcf-project>)
  add_test(NAME project_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/contract.py $<TARGET_FILE:kcf-project>)
endif()

set(KCF_PM_TEST_SDK "" CACHE PATH "Installed KCF SDK for real integration tests")
if(BUILD_TESTING AND KCF_PM_TEST_SDK)
  add_test(NAME sdk_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:kcf-project> ${KCF_PM_TEST_SDK})
  set_tests_properties(sdk_integration PROPERTIES TIMEOUT 240)
  if(KCF_PM_GUI)
    add_test(NAME qt_workflow COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/gui.py $<TARGET_FILE:kcf-project-manager> ${KCF_PM_TEST_SDK})
    set_tests_properties(qt_workflow PROPERTIES TIMEOUT 180)
  endif()
endif()
