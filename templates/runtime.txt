#pragma once
#include <nlohmann/json.hpp>
#include <filesystem>
#include <fstream>
#include <iostream>
#include <cstdlib>
#include <cmath>
#include <unistd.h>
namespace pm_runtime {
using Json = nlohmann::json;
namespace fs = std::filesystem;
inline Json read(const fs::path& path) {
    std::ifstream input(path);
    if (!input) throw std::runtime_error("Cannot read " + path.string());
    Json j; input >> j; return j;
}
inline Json project(const fs::path& path) {
    auto j = read(path);
    if (j.at("schema") != 1 || !j.at("elements").is_array())
        throw std::runtime_error("Unsupported project schema");
    return j;
}
inline double hz(const Json& e) {
    double v = e.at("hz").get<double>();
    if (!std::isfinite(v) || v <= 0 || v > 100000) throw std::runtime_error("Invalid Hz");
    return v;
}
inline void environment(const Json& e, const fs::path& root) {
    for (auto it = e.at("env").begin(); it != e.at("env").end(); ++it) {
        if (it.key().empty() || it.key().find('=') != std::string::npos || it.key().rfind("KCF_", 0) == 0)
            throw std::runtime_error("Invalid/reserved environment variable");
        if (::setenv(it.key().c_str(), it.value().get<std::string>().c_str(), 1))
            throw std::runtime_error("setenv failed");
    }
    fs::path cwd = e.at("workdir").get<std::string>();
    fs::current_path(cwd.is_absolute() ? cwd : root / cwd);
}
inline std::map<std::string, std::string> artifacts(const fs::path& path) {
    std::ifstream input(path);
    if (!input) throw std::runtime_error("Cannot read CMake artifact metadata: " + path.string());
    std::map<std::string, std::string> result;
    std::string target, binary;
    while (std::getline(input, target)) {
        if (!std::getline(input, binary) || !fs::is_regular_file(binary) || access(binary.c_str(), X_OK))
            throw std::runtime_error("Missing executable for " + target + "; build the target first");
        if (!result.emplace(target, binary).second) throw std::runtime_error("Duplicate target metadata");
    }
    return result;
}
}
